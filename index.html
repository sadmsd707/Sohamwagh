// --- Physics Drawing Routines --- function drawResistor(ctx, x, y, size) { const w = size * 1.8; ctx.beginPath(); ctx.moveTo(x, y); ctx.lineTo(x + w * 0.2, y); // Lead-in line // Zigzag points ctx.lineTo(x + w * 0.3, y - 10); ctx.lineTo(x + w * 0.4, y + 10); ctx.lineTo(x + w * 0.5, y - 10); ctx.lineTo(x + w * 0.6, y + 10); ctx.lineTo(x + w * 0.7, y - 10); ctx.lineTo(x + w * 0.8, y); ctx.lineTo(x + w, y); // Lead-out line ctx.stroke(); ctx.font = "bold 10px monospace"; ctx.fillText("R", x + w*0.45, y - 14); } function drawCapacitor(ctx, x, y, size) { const w = size * 1.2; const h = 16; ctx.beginPath(); // Left lead and plate ctx.moveTo(x, y); ctx.lineTo(x + w * 0.4, y); ctx.moveTo(x + w * 0.4, y - h); ctx.lineTo(x + w * 0.4, y + h); // Right plate and lead ctx.moveTo(x + w * 0.6, y - h); ctx.lineTo(x + w * 0.6, y + h); ctx.moveTo(x + w * 0.6, y); ctx.lineTo(x + w, y); ctx.stroke(); ctx.font = "bold 10px monospace"; ctx.fillText("C", x + w*0.42, y - h - 4); } function drawDiode(ctx, x, y, size) { const w = size * 1.4; const h = 12; ctx.beginPath(); // Entrance Line ctx.moveTo(x, y); ctx.lineTo(x + w * 0.3, y); // Triangle arrowhead pointer ctx.moveTo(x + w * 0.3, y - h); ctx.lineTo(x + w * 0.7, y); ctx.lineTo(x + w * 0.3, y + h); ctx.closePath(); // Vertical Block line ctx.moveTo(x + w * 0.7, y - h); ctx.lineTo(x + w * 0.7, y + h); // Exit Line ctx.moveTo(x + w * 0.7, y); ctx.lineTo(x + w, y); ctx.stroke(); ctx.font = "bold 9px monospace"; ctx.fillText("D", x + w*0.45, y - h - 4); } class ScienceParticle { constructor() { this.reset(); this.y = Math.random() * height; } reset() { // Fair split logic across all branches: // 50% Text Expressions, 25% Chemistry Structures, 25% Physics Circuit Elements const rand = Math.random(); if (rand < 0.5) { this.type = 'text'; } else if (rand < 0.75) { this.type = 'chemistry'; this.subType = Math.random() > 0.5 ? 'benzene' : 'chain'; } else { this.type = 'physics'; const roll = Math.random(); this.subType = roll < 0.33 ? 'resistor' : (roll < 0.66 ? 'capacitor' : 'diode'); } this.text = expressions[Math.floor(Math.random() * expressions.length)]; this.x = Math.random() * width; this.y = -50; this.speed = 0.2 + Math.random() * 0.4; this.size = 13 + Math.floor(Math.random() * 15); this.opacity = 0.18 + Math.random() * 0.14; } update() { this.y += this.speed; if (this.y > height + 50) { this.reset(); } } draw() { ctx.save(); ctx.strokeStyle = `rgba(157, 23, 77, ${this.opacity})`; ctx.fillStyle = `rgba(157, 23, 77, ${this.opacity})`; ctx.lineWidth = 1.8; if (this.type === 'text') { ctx.font = `bold ${this.size}px 'Courier New', monospace`; ctx.fillText(this.text, this.x, this.y); } else if (this.type === 'chemistry') { if (this.subType === 'benzene') { drawBenzene(ctx, this.x, this.y, this.size * 1.2); } else { drawSkeletalChain(ctx, this.x, this.y, this.size * 1.5); } } else if (this.type === 'physics') { if (this.subType === 'resistor') { drawResistor(ctx, this.x, this.y, this.size); } else if (this.subType === 'capacitor') { drawCapacitor(ctx, this.x, this.y, this.size); } else if (this.subType === 'diode') { drawDiode(ctx, this.x, this.y, this.size); } } ctx.restore(); } } const particleCount = 45;
